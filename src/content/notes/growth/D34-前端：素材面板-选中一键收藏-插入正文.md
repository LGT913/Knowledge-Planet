## 6. D34 前端：素材面板 + 选中一键收藏 + 插入正文

### 6.1 代码

**`electron/src/api/material.ts`**（新建）

```ts
import http from './http';

export interface Material {
    id: number; genre: string; kind: string; content: string;
    source: string; createdAt: string; updatedAt: string;
}

export const materialApi = {
    list: (genre?: string) =>
        http.get<Material[]>('/api/materials', { params: { genre } }).then(r => r.data.data),
    create: (m: Partial<Material>) => http.post<Material>('/api/materials', m).then(r => r.data.data),
    update: (id: number, m: Partial<Material>) =>
        http.put<Material>(`/api/materials/${id}`, m).then(r => r.data.data),
    remove: (id: number) => http.delete(`/api/materials/${id}`),
};
```

<template>

<script setup lang="ts">
import { reactive, ref } from 'vue';
import { ElMessage } from 'element-plus';
import { useEditorStore } from '../stores/editor';
import { materialApi, type Material } from '../api/material';

const props = defineProps<{
    /** 打开面板时可预填的素材内容（"选中收藏"场景由写作页注入） */
    presetContent?: string;
}>();
const visible = defineModel<boolean>({ required: true });
const editorStore = useEditorStore();

const GENRES = ['NOVEL', 'MEDIA', 'OFFICIAL', 'CUSTOM'];
const LABELS: Record<string, string> = { NOVEL: '网文', MEDIA: '自媒体', OFFICIAL: '公文', CUSTOM: '自定义' };
const KIND_LABELS: Record<string, string> = { QUOTE: '金句', CASE: '案例', PHRASE: '专业表述' };

const genreFilter = ref('');
const materials = ref<Material[]>([]);
const dialogOpen = ref(false);
const form = reactive<Partial<Material>>({});

async function reload() {
    materials.value = await materialApi.list(genreFilter.value || undefined);
}

/** 面板打开：若有预填内容（选中收藏），自动弹出收藏对话框 */
function onOpen() {
    reload().then(() => {
        if (props.presetContent) openDialog();
    });
}

function openDialog(m?: Material) {
    form.id = m?.id; form.genre = m?.genre ?? 'NOVEL'; form.kind = m?.kind ?? 'QUOTE';
    form.content = m?.content ?? props.presetContent ?? '';
    form.source = m?.source ?? '';
    dialogOpen.value = true;
}

async function saveDialog() {
    if (!form.content?.trim()) { ElMessage.warning('素材内容不能为空'); return; }
    if (form.id) await materialApi.update(form.id, form);
    else await materialApi.create(form);
    dialogOpen.value = false;
    await reload();
    ElMessage.success('已收藏');
}

async function onDelete(id: number) {
    await materialApi.remove(id);
    await reload();
}

/** 插入光标处：原样粘贴 content（纯文本），TipTap 事务进撤销栈 */
function insert(text: string) {
    const ed = editorStore.editor;
    if (!ed) return;
    ed.chain().focus().insertContent(text).run();
    ElMessage.success('已插入光标处');
}
</script>

```ts
import { ref } from 'vue';
import OutlinePanel from '../components/OutlinePanel.vue';
import InspirePanel from '../components/InspirePanel.vue';
import MaterialPanel from '../components/MaterialPanel.vue';
import TemplatePanel from '../components/TemplatePanel.vue';
import { useEditorStore } from '../stores/editor';

const editorStore = useEditorStore();
const outlineOpen = ref(false);
const inspireOpen = ref(false);
const materialOpen = ref(false);
const templateOpen = ref(false);
const pendingMaterial = ref<string | undefined>(undefined);

/** 选中文本一键收藏：取选区纯文本预填素材面板 */
function collectSelection() {
    const ed = editorStore.editor;
    if (!ed) return;
    const { from, to } = ed.state.selection;
    if (to <= from) { ElMessage.warning('请先选中要收藏的文字'); return; }
    pendingMaterial.value = ed.state.doc.textBetween(from, to);
    materialOpen.value = true;
}
```

> 工具栏可加 `@click="collectSelection"` 的【📌 收藏选中】按钮（有选区才可用，与 W3 quick 的 `hasSelection` 同判断）。
> 拖拽插入说明：v1.0 以"点击插入"为主；面板卡片可加 `draggable` + `dragstart` 存文本、编辑器 `drop` 拦截 `insertText` 作为增强（大纲拖拽排序已明确裁剪，素材拖拽同理可选做，不阻塞验收）。
