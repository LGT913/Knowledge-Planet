## 2. D30 前端：文体切换工具栏 + 模板面板 + 设置页重载

### 2.1 任务

- 工具栏文体切换（网文/自媒体/公文/CUSTOM）：切换后下一轮 AI 调用携带对应总纲（后端 styleSystem 已按 genre 注入）；当前文体记忆 localStorage（02 §4 `currentGenre`）；切换同步更新文档 meta（genre），**不影响正文**。
- 模板面板：预制（只读）+ 自定义（增删改），按 genre 分栏。
- 设置页新增【重载提示词配置】按钮。

### 2.2 代码

**`electron/src/stores/editor.ts`** 修改（文体持久化）：

```ts
import { defineStore } from 'pinia';
import type { Editor } from '@tiptap/core';

const GENRE_KEY = 'quill-current-genre';
const GENRES = ['NOVEL', 'MEDIA', 'OFFICIAL', 'CUSTOM'];

export const useEditorStore = defineStore('editor', {
    state: () => ({
        editor: null as Editor | null,
        genre: (GENRES.includes(localStorage.getItem(GENRE_KEY) ?? '') ? localStorage.getItem(GENRE_KEY) : 'NOVEL') as string,
    }),
    actions: {
        setEditor(e: Editor | null) { this.editor = e; },
        setGenre(g: string) {
            if (!GENRES.includes(g)) return;
            this.genre = g;
            localStorage.setItem(GENRE_KEY, g);   // 02-接口文档 §4 currentGenre
        },
    },
});
```

**`electron/src/api/template.ts`**（新建）

```ts
import http from './http';

export interface PromptTemplate {
    id: number; name: string; prompt: string; genre: string;
    category: string; isSystem: boolean; createdAt: string;
}

export const templateApi = {
    list: (genre?: string) =>
        http.get<PromptTemplate[]>('/api/prompt-templates', { params: { genre } }).then(r => r.data.data),
    upsert: (t: Partial<PromptTemplate>) =>
        http.post<PromptTemplate>('/api/prompt-templates', t).then(r => r.data.data),
    remove: (id: number) => http.delete(`/api/prompt-templates/${id}`),
    reload: () => http.post('/api/prompt-templates/reload'),
};
```

    ElMessage.success('已复制到剪贴板');
}

import { useEditorStore } from '../stores/editor';
const editorStore = useEditorStore();

/** 切换文体：只更新会话文体 + 文档 meta，不动正文；下一轮 AI 调用即携带新总纲 */
